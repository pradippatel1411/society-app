Step 1 — Product owner logs inpowershell
# Send OTP to your owner mobile
$body = @{ mobile="8734032255"; scope="owner" } | ConvertTo-Json
Invoke-RestMethod -Uri "http://localhost:8787/auth/sendOTP" -Method Post -Body $body -ContentType "application/json"


# Verify OTP and capture the owner token
$body = @{ mobile="8734032255"; otp="000000"; scope="owner" } | ConvertTo-Json
$ownerLogin = Invoke-RestMethod -Uri "http://localhost:8787/auth/verifyOTP" -Method Post -Body $body -ContentType "application/json"
$ownerToken = $ownerLogin.token
$ownerHeaders = @{ Authorization = "Bearer $ownerToken" }

Write-Host "Owner logged in: $($ownerLogin.user.userType)" -ForegroundColor Green
